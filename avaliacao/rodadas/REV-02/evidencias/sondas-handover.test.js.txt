/* global beforeAll, afterEach, describe, it, expect, jest */
/* Sondas de avaliação: as expectativas são as do contrato, inclusive quando falham. */
const { db, insertSector, insertSchool, insertClass } = require('../../../../tests/auxiliarFunctions');
const request = require('supertest');
const app = require('../../../../src/app');

afterEach(() => jest.restoreAllMocks());

beforeAll(async () => {
  const result = await db.query('SELECT current_database() AS banco, current_schema() AS schema, version() AS versao');
  expect(result.rows[0].schema).toMatch(/^eleva_test_\d+_[a-f0-9]+$/);
  console.log('AMBIENTE_ISOLADO ' + JSON.stringify(result.rows[0]));
});

const asJson = (value) => JSON.parse(JSON.stringify(value));
const classBody = (changes = {}) => ({
  name: 'turma de sonda', segment: 'Ensino Fundamental', grade: '1º ano',
  shift: 'Matutino', status: 'current', maxOccupancy: 20, ...changes,
});

function record (id, input, expected, observed) {
  console.log('OBSERVACAO ' + JSON.stringify({ id, input, expected, observed }));
}

describe('REV-02 — complementos do contrato das operações do handover', () => {
  it.each([
    ['name', 'a'.repeat(100)], ['address', 'a'.repeat(255)], ['headteacher', 'a'.repeat(150)],
    ['phone', '+552223456789'], ['phone', '+5522998765432'],
  ])('S01: PATCH de escola aceita o limite/formato exato de %s: %p', async (field, value) => {
    const sector = await insertSector();
    const school = await insertSchool(sector.id);
    const response = await request(app).patch(`/schools/${school.id}`).send({ [field]: value });
    const stored = asJson((await db.query('SELECT * FROM schools WHERE id = $1', [school.id])).rows[0]);
    expect(response.status).toBe(200);
    expect(response.body[field]).toBe(value);
    expect(stored).toEqual(response.body);
    expect(stored.createdAt).toBe(school.createdAt);
  });

  it.each([
    ['name', false], ['address', {}], ['headteacher', []], ['phone', false], ['sectorId', '12abc'],
  ])('S02: PATCH de escola rejeita tipo inválido de %s sem gravar: %p', async (field, value) => {
    const sector = await insertSector();
    const school = await insertSchool(sector.id);
    const response = await request(app).patch(`/schools/${school.id}`).send({ [field]: value });
    const stored = asJson((await db.query('SELECT * FROM schools WHERE id = $1', [school.id])).rows[0]);
    expect(response.status).toBe(422);
    expect(response.body.message).toContain(field);
    expect(stored).toEqual(school);
  });

  it.each([['0', 0], ['2.0', 2], ['2e1', 20]])(
    'S03: cadastro de turma converte capacidade textual %s para %s', async (input, expected) => {
      const sector = await insertSector();
      const school = await insertSchool(sector.id);
      const response = await request(app).post(`/schools/${school.id}/classes`).send(classBody({ maxOccupancy: input }));
      expect(response.status).toBe(201);
      expect(response.body.maxOccupancy).toBe(expected);
      const stored = asJson((await db.query('SELECT * FROM classes WHERE id = $1', [response.body.id])).rows[0]);
      expect(stored).toEqual(response.body);
    },
  );

  const approvedCombinations = [
    ['Educação Infantil — Creche', 'Berçário'], ['Educação Infantil — Creche', 'Maternal I'],
    ['Educação Infantil — Creche', 'Maternal II'], ['Educação Infantil — Pré-escola', 'Jardim I'],
    ['Educação Infantil — Pré-escola', 'Jardim II'],
    ...['1º ano', '2º ano', '3º ano', '4º ano', '5º ano', '6º ano', '7º ano', '8º ano', '9º ano']
      .map((grade) => ['Ensino Fundamental', grade]),
    ...['1º ano', '2º ano', '3º ano'].map((grade) => ['Ensino Médio', grade]),
  ];
  it.each(approvedCombinations)('S04: PATCH de turma aceita a combinação aprovada %s / %s', async (segment, grade) => {
    const sector = await insertSector();
    const school = await insertSchool(sector.id);
    const current = await insertClass(school.id, { status: 'completed', createdAt: '2020-01-01T00:00:00.000Z' });
    const response = await request(app).patch(`/classes/${current.id}`).send({ segment, grade });
    const stored = asJson((await db.query('SELECT * FROM classes WHERE id = $1', [current.id])).rows[0]);
    expect(response.status).toBe(200);
    expect(response.body).toEqual({ ...current, segment, grade, updatedAt: expect.any(String) });
    expect(stored).toEqual(response.body);
  });

  it('S05: detalhes refletem a mudança de setor da escola sem alterar vínculo nem datas da turma', async () => {
    const origin = await insertSector();
    const destination = await insertSector();
    const school = await insertSchool(origin.id);
    const current = await insertClass(school.id);
    await request(app).patch(`/schools/${school.id}`).send({ sectorId: destination.id }).expect(200);
    const response = await request(app).get(`/classes/${current.id}`).expect(200);
    expect(response.body).toEqual({ ...current, school: { id: school.id, name: school.name } });
    expect(asJson((await db.query('SELECT * FROM classes WHERE id = $1', [current.id])).rows[0])).toEqual(current);
  });
});

describe('REV-02 — rejeição de campos desconhecidos antes da sanitização', () => {
  const rawBody = (top, inherited) => JSON.stringify({ ...top, ['__proto__']: inherited });

  it('S06-controle: campo desconhecido comum é rejeitado no PATCH sem alterar a turma', async () => {
    const sector = await insertSector();
    const school = await insertSchool(sector.id);
    const current = await insertClass(school.id);
    const input = { unexpected: { status: 'completed', maxOccupancy: 0 } };
    const response = await request(app).patch(`/classes/${current.id}`).send(input);
    const stored = asJson((await db.query('SELECT * FROM classes WHERE id = $1', [current.id])).rows[0]);
    record('S06-controle', input, { status: 422, sem_mutacao: true }, { status: response.status, body: response.body, stored });
    expect(response.status).toBe(422);
    expect(stored).toEqual(current);
  });

  it('S06-escola: __proto__ deve ser rejeitado sem aplicar o sectorId aninhado', async () => {
    const origin = await insertSector();
    const destination = await insertSector();
    const school = await insertSchool(origin.id);
    const raw = rawBody({ address: 'Rua alterada pela sonda' }, { sectorId: destination.id });
    const response = await request(app).patch(`/schools/${school.id}`)
      .set('Content-Type', 'application/json').send(raw);
    const stored = asJson((await db.query('SELECT * FROM schools WHERE id = $1', [school.id])).rows[0]);
    record('S06-escola', JSON.parse(raw), { status: 422, stored: school }, { status: response.status, body: response.body, stored });
    expect({ status: response.status, stored }).toEqual({ status: 422, stored: school });
  });

  it('S06-cadastro: __proto__ não substitui os seis campos obrigatórios do cadastro de turma', async () => {
    const sector = await insertSector();
    const school = await insertSchool(sector.id);
    const raw = rawBody({}, classBody());
    const response = await request(app).post(`/schools/${school.id}/classes`)
      .set('Content-Type', 'application/json').send(raw);
    const stored = asJson((await db.query('SELECT * FROM classes ORDER BY id')).rows);
    record('S06-cadastro', JSON.parse(raw), { status: 422, stored: [] }, { status: response.status, body: response.body, stored });
    expect({ status: response.status, stored }).toEqual({ status: 422, stored: [] });
  });

  it('S06-turma: __proto__ deve ser rejeitado sem aplicar status e capacidade aninhados', async () => {
    const sector = await insertSector();
    const school = await insertSchool(sector.id);
    const current = await insertClass(school.id);
    const raw = rawBody({}, { status: 'completed', maxOccupancy: 0 });
    const response = await request(app).patch(`/classes/${current.id}`)
      .set('Content-Type', 'application/json').send(raw);
    const stored = asJson((await db.query('SELECT * FROM classes WHERE id = $1', [current.id])).rows[0]);
    record('S06-turma', JSON.parse(raw), { status: 422, stored: current }, { status: response.status, body: response.body, stored });
    expect({ status: response.status, stored }).toEqual({ status: 422, stored: current });
  });
});
